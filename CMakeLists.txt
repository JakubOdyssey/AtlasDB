cmake_minimum_required(VERSION 3.24)
project(AtlasDB VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(GNUInstallDirs)
option(ATLAS_ASAN "Enable AddressSanitizer" OFF)
option(ATLAS_UBSAN "Enable UndefinedBehaviorSanitizer" OFF)
option(ATLAS_TSAN "Enable ThreadSanitizer" OFF)
option(ATLAS_CLANG_TIDY "Run clang-tidy" OFF)
if(ATLAS_TSAN AND (ATLAS_ASAN OR ATLAS_UBSAN))
  message(FATAL_ERROR "TSan must use its own build")
endif()
if(MSVC AND (ATLAS_UBSAN OR ATLAS_TSAN))
  message(FATAL_ERROR "UBSan/TSan require a supported Unix Clang or GCC toolchain")
endif()
if(ATLAS_CLANG_TIDY)
  find_program(CLANG_TIDY_EXECUTABLE clang-tidy REQUIRED)
  set(CMAKE_CXX_CLANG_TIDY ${CLANG_TIDY_EXECUTABLE})
endif()
add_library(atlas_options INTERFACE)
target_compile_features(atlas_options INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(atlas_options INTERFACE /W4 /permissive- /EHsc /utf-8)
  target_compile_definitions(atlas_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(ATLAS_ASAN)
    string(REPLACE "/RTC1" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
    target_compile_options(atlas_options INTERFACE /fsanitize=address /Zi)
    target_link_options(atlas_options INTERFACE /INCREMENTAL:NO)
  endif()
else()
  target_compile_options(atlas_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  foreach(san ASAN UBSAN TSAN)
    if(ATLAS_${san})
      if(san STREQUAL "ASAN")
        set(flag address)
      elseif(san STREQUAL "UBSAN")
        set(flag undefined)
        target_compile_options(atlas_options INTERFACE -fno-sanitize-recover=undefined)
      else()
        set(flag thread)
      endif()
      target_compile_options(atlas_options INTERFACE -fsanitize=${flag} -fno-omit-frame-pointer)
      target_link_options(atlas_options INTERFACE -fsanitize=${flag})
    endif()
  endforeach()
endif()
find_package(Threads REQUIRED)
add_library(atlas
  src/page.cpp src/slotted_page.cpp src/file.cpp src/disk_manager.cpp
  src/buffer_pool.cpp src/bplus_tree.cpp src/wal.cpp src/database.cpp)
add_library(atlas::atlas ALIAS atlas)
target_include_directories(atlas PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(atlas PUBLIC atlas_options Threads::Threads)
add_executable(atlasdb tools/atlasdb/main.cpp)
target_link_libraries(atlasdb PRIVATE atlas)
add_executable(atlas-chaos tools/atlas-chaos/main.cpp)
target_link_libraries(atlas-chaos PRIVATE atlas)
add_executable(atlas-bench benchmarks/main.cpp)
target_link_libraries(atlas-bench PRIVATE atlas)
add_executable(atlas-example examples/basic.cpp)
target_link_libraries(atlas-example PRIVATE atlas)
if(BUILD_TESTING)
  set(ATLAS_TEST_TIMEOUT 240 CACHE STRING "Timeout per test group in seconds")
  set(ATLAS_CHAOS_TIMEOUT 300 CACHE STRING "Timeout for the CTest chaos campaign in seconds")
  add_executable(atlas-tests tests/main.cpp tests/unit/storage.cpp tests/integration/database.cpp tests/property/model.cpp tests/recovery/wal.cpp)
  target_link_libraries(atlas-tests PRIVATE atlas)
  target_include_directories(atlas-tests PRIVATE tests)
  foreach(group unit integration property recovery)
    add_test(NAME ${group} COMMAND atlas-tests ${group})
    set_tests_properties(${group} PROPERTIES LABELS ${group} TIMEOUT ${ATLAS_TEST_TIMEOUT})
  endforeach()
  add_test(NAME chaos COMMAND atlas-chaos --iterations 40 --seed 20260920)
  set_tests_properties(chaos PROPERTIES LABELS "chaos;recovery" TIMEOUT ${ATLAS_CHAOS_TIMEOUT})
  if(MSVC AND ATLAS_ASAN)
    get_filename_component(ATLAS_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set_tests_properties(unit integration property recovery chaos PROPERTIES
      ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${ATLAS_COMPILER_DIR}")
  endif()
endif()
install(TARGETS atlas atlas_options EXPORT AtlasTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT AtlasTargets NAMESPACE atlas:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AtlasDB)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/AtlasDBConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/AtlasDBConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AtlasDB)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/AtlasDBConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/AtlasDBConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/AtlasDBConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AtlasDB)
